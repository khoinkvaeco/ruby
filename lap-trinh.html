1. Chương trình đầu tiên

Lệnh print() in ra màn hình. Dòng bắt đầu bằng # là chú thích, máy tính bỏ qua.

# Chương trình đầu tiên của Minh Khuê
print("Xin chào, mình là Minh Khuê!")
print("Năm nay mình học lớp", 9)
✏️ Bài tập: in ra 3 dòng giới thiệu bản thân (tên, môn học yêu thích, sở thích). 
print("Tên: Minh Khuê")
print("Môn yêu thích: Toán")
print("Sở thích: đọc sách, lập trình")

3. Nhập dữ liệu & phép toán

input() luôn trả về chuỗi → muốn tính toán phải đổi sang số bằng int() hoặc float().

Phép toán: + - * /, chia lấy nguyên //, chia lấy dư %, luỹ thừa **.

a = float(input("Nhập cạnh thứ nhất: "))
b = float(input("Nhập cạnh thứ hai: "))
c = (a**2 + b**2) ** 0.5          # định lí Pythagore
print("Cạnh huyền =", round(c, 2))
print("17 // 5 =", 17 // 5, "; 17 % 5 =", 17 % 5)
✏️ Bài tập: nhập bán kính r, in ra chu vi và diện tích hình tròn. 
import math
r = float(input("Bán kính: "))
print("Chu vi:", round(2 * math.pi * r, 2))
print("Diện tích:", round(math.pi * r ** 2, 2))

4. Rẽ nhánh if – elif – else

Python dùng thụt lề (4 dấu cách) để biết lệnh nào thuộc khối nào. So sánh: == != < > <= >=; kết hợp: and, or, not.

diem = float(input("Nhập điểm trung bình: "))
if diem >= 8:
    print("Học lực Tốt 🌟")
elif diem >= 6.5:
    print("Học lực Khá 👍")
elif diem >= 5:
    print("Học lực Đạt")
else:
    print("Cần cố gắng thêm 💪")
✏️ Bài tập: nhập một năm, cho biết có phải năm nhuận không (chia hết cho 400, hoặc chia hết cho 4 nhưng không chia hết cho 100). 
nam = int(input("Năm: "))
if nam % 400 == 0 or (nam % 4 == 0 and nam % 100 != 0):
    print(nam, "là năm nhuận")
else:
    print(nam, "không phải năm nhuận")

5. Vòng lặp for, while

for khi biết trước số lần lặp; while lặp khi điều kiện còn đúng. range(a, b) tạo dãy a, a+1, …, b−1.

# Bảng cửu chương 7
for i in range(1, 11):
    print(f"7 x {i} = {7 * i}")

# Tổng các số từ 1 đến 100
tong = 0
for so in range(1, 101):
    tong += so
print("Tổng 1..100 =", tong)

# while: đếm số chữ số của một số
n = 2026
dem = 0
while n > 0:
    n //= 10
    dem += 1
print("Số chữ số:", dem)
✏️ Bài tập: in các số nguyên tố nhỏ hơn 50. 
for n in range(2, 50):
    la_nguyen_to = True
    for d in range(2, int(n ** 0.5) + 1):
        if n % d == 0:
            la_nguyen_to = False
            break
    if la_nguyen_to:
        print(n, end=" ")

6. Danh sách (list)

diem = [8.5, 9, 7.25, 10, 6.5]
print("Điểm đầu tiên:", diem[0])        # chỉ số bắt đầu từ 0
print("Điểm cuối:", diem[-1])
diem.append(9.5)                        # thêm vào cuối
print("Số bài:", len(diem))
print("Cao nhất:", max(diem), "- Thấp nhất:", min(diem))
print("Trung bình:", round(sum(diem) / len(diem), 2))
print("Sắp xếp:", sorted(diem))
for d in diem:
    if d >= 9:
        print("Giỏi quá:", d)
✏️ Bài tập: đếm có bao nhiêu điểm dưới 8 trong danh sách trên. 
diem = [8.5, 9, 7.25, 10, 6.5, 9.5]
dem = 0
for d in diem:
    if d < 8:
        dem += 1
print("Số điểm dưới 8:", dem)

7. Chuỗi (string)

cau = "Minh Khuê học lập trình"
print(len(cau))              # độ dài
print(cau.upper())           # viết hoa
print(cau.split())           # tách thành các từ

print("Khuê" in cau)         # kiểm tra chứa
print(cau.replace("lập trình", "Python"))
print("ana"[::-1])           # đảo ngược chuỗi
✏️ Bài tập: kiểm tra một từ có phải "palindrome" (đọc xuôi ngược như nhau) không. 
tu = input("Nhập một từ: ").lower()
if tu == tu[::-1]:
    print("Là palindrome!")
else:
    print("Không phải palindrome")

8. Hàm (function)

Hàm gom một đoạn code để dùng lại nhiều lần. def để định nghĩa, return để trả kết quả.

def dien_tich_tam_giac(day, cao):
    return day * cao / 2

def xep_loai(diem):
    if diem >= 8:
        return "Tốt"
    elif diem >= 6.5:
        return "Khá"
    elif diem >= 5:
        return "Đạt"
    return "Chưa đạt"

print(dien_tich_tam_giac(6, 4))
for d in [9, 7, 4.5]:
    print(d, "→", xep_loai(d))
✏️ Bài tập: viết hàm ucln(a, b) tìm ước chung lớn nhất bằng thuật toán Euclid. 
def ucln(a, b):
    while b != 0:
        a, b = b, a % b
    return a

print(ucln(48, 36))   # 12

📐 Giải phương trình bậc hai (kết hợp Toán 9!)

import math
a = float(input("a = "))
b = float(input("b = "))
c = float(input("c = "))
if a == 0:
    print("Không phải phương trình bậc hai")
else:
    delta = b * b - 4 * a * c
    print("Δ =", delta)
    if delta > 0:
        x1 = (-b + math.sqrt(delta)) / (2 * a)
        x2 = (-b - math.sqrt(delta)) / (2 * a)
        print("Hai nghiệm phân biệt:", x1, x2)
    elif delta == 0:
        print("Nghiệm kép: x =", -b / (2 * a))
    else:
        print("Phương trình vô nghiệm")

11. Học gì tiếp theo?

  • Thuật toán: tìm kiếm tuần tự / nhị phân, sắp xếp (nổi bọt, chọn), đệ quy — nền tảng thi học sinh giỏi Tin. Luyện đề tại Codeforces, VNOI.
  • Web: HTML (khung), CSS (trang trí), JavaScript (tương tác) — freeCodeCamp, MDN.
  • Trò chơi: thư viện pygame trong Python, hoặc Scratch nâng cao.
  • Khoa học máy tính: khoá CS50 (Harvard, miễn phí, có phụ đề).
  • Trí tuệ nhân tạo: khi đã vững Python, thử Kaggle Learn.
🤖 Dùng AI như một gia sư: hỏi "vì sao code này lỗi", "giải thích dòng này" — đừng nhờ viết hộ cả bài, vì như vậy sẽ không học được gì.

Bài 6. Tính tổng các chữ số của một số nguyên dương (ví dụ 2026 → 10).

Bài 7. Cho danh sách điểm, in điểm cao nhất, thấp nhất và các điểm trên trung bình — không dùng max(), min().
Xem lời giải
diem = [7, 9.5, 6, 8, 10, 5.5]
cao = thap = diem[0]
for d in diem:
    if d > cao:
        cao = d
    if d < thap:
        thap = d
tb = sum(diem) / len(diem)
print("Cao nhất:", cao, "- Thấp nhất:", thap, "- TB:", round(tb, 2))
print("Trên trung bình:", [d for d in diem if d > tb])

Bài 10. Viết hàm la_nguyen_to(n) và dùng nó in mọi số nguyên tố nhỏ hơn 100, kèm số lượng.
Xem lời giải
def la_nguyen_to(n):
    if n < 2:
        return False
    d = 2
    while d * d <= n:
        if n % d == 0:
            return False
        d += 1
    return True

ds = [n for n in range(100) if la_nguyen_to(n)]
print(ds)
print("Có", len(ds), "số nguyên tố nhỏ hơn 100")
Bài 11. Sắp xếp danh sách tăng dần bằng thuật toán sắp xếp nổi bọt (không dùng sort()).
Xem lời giải
a = [5, 2, 9, 1, 7, 3]
n = len(a)
for i in range(n - 1):
    for j in range(n - 1 - i):
        if a[j] > a[j + 1]:
            a[j], a[j + 1] = a[j + 1], a[j]   # đổi chỗ
print(a)
Bài 12. Tìm kiếm nhị phân: trong danh sách đã sắp xếp, tìm vị trí của một số, đếm số bước — so sánh với tìm tuần tự.
Xem lời giải
a = list(range(1, 1001, 3))   # 1, 4, 7, ..., 1000 (đã sắp xếp)
x = 703
trai, phai, buoc = 0, len(a) - 1, 0
while trai <= phai:
    buoc += 1
    giua = (trai + phai) // 2
    if a[giua] == x:
        print("Tìm thấy", x, "ở vị trí", giua, "sau", buoc, "bước")
        break
    elif a[giua] < x:
        trai = giua + 1
    else:
        phai = giua - 1
else:
    print("Không có", x)
print("Tìm tuần tự cần khoảng", a.index(x) + 1, "bước")
